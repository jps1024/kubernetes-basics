
«« PART 1 »»
«« Standard MariaDB Helm chart deployment in replication mode »»

helm repo add bitnami https://charts.bitnami.com/bitnami

helm show chart bitnami/mariadb --version 22.0.0
helm show values bitnami/mariadb --version 22.0.0 > values-22.0.0.yaml

helm pull bitnami/mariadb --version 22.0.0
helm pull bitnami/mariadb --version 22.0.0 --untar

./load-images.sh  # optional
helm install mymaria bitnami/mariadb --version 22.0.0 -f values-22.0.0-mymaria.yaml -n mariadb-system --create-namespace
helm install mymaria bitnami/mariadb --version 22.0.0 -f values-22.0.0-mymaria-mini.yaml -n mariadb-system --create-namespace
helm upgrade mymaria bitnami/mariadb --version 22.0.0 -f values-22.0.0-mymaria-mini.yaml -n mariadb-system

kc config set-context --current --namespace=mariadb-system
helm history mymaria

helm delete mymaria -n mariadb-system
kc delete ns mariadb-system

kubectl run mymaria-client-0 --rm --tty -i                                \
            --restart='Never'                                             \
            --image docker.io/bitnamilegacy/mariadb:12.0.2-debian-12-r0   \
            --namespace mariadb-system                                    \
            --command -- mysql -uroot -ppasswd878 -h mymaria-mariadb-primary database878

kubectl run mymaria-client-1 --rm --tty -i                                \
            --restart='Never'                                             \
            --image docker.io/bitnamilegacy/mariadb:12.0.2-debian-12-r0   \
            --namespace mariadb-system                                    \
            --command -- mysql -uuser878 -ppasswd878 -h mymaria-mariadb-secondary database878

./insert-data.sh | kubectl run mymaria-inserter --rm -i                   \
            --restart='Never'                                             \
            --image docker.io/bitnamilegacy/mariadb:12.0.2-debian-12-r0   \
            --namespace mariadb-system                                    \
            --command -- mysql -uuser878 -ppasswd878 -h mymaria-mariadb-primary database878

echo "SELECT COUNT(*) from TBL01;" | kubectl run mymaria-query --rm -i                        \
            --restart='Never'                                                                 \
            --image docker.io/bitnamilegacy/mariadb:12.0.2-debian-12-r0                       \
            --namespace mariadb-system                                                        \
            --command -- mysql -uuser878 -ppasswd878 -h mymaria-mariadb-primary database878

echo "select COL01,COL02,COL03,COL04 from TBL01 where COL01=\"81446e1b9e\";" | kubectl run mymaria-query --rm -i  \
            --restart='Never'                                                                                     \
            --image docker.io/bitnamilegacy/mariadb:12.0.2-debian-12-r0                                           \
            --namespace mariadb-system                                                                            \
            --command -- mysql -uuser878 -ppasswd878 -h mymaria-mariadb-primary database878

SQL queries:
    USE database878;
    SHOW TABLES;
    SELECT * FROM TBL01 LIMIT 20;
    DESCRIBE TBL01;
    SELECT COUNT(*) from TBL01;
    DROP TABLE TBL01;


«« PART 2 »»
«« Addition of metrics generation, ready to be scraped by Prometheus »»

Option 1 (losing all existing DB data):
    helm delete mymaria -n mariadb-system
    kc delete ns mariadb-system
    helm install mymaria bitnami/mariadb --version 22.0.0 -f values-22.0.0-mymaria-mini-prometheus.yaml -n mariadb-system --create-namespace
Or option2 (keeping DB data intact):
    helm upgrade mymaria bitnami/mariadb --version 22.0.0 -f values-22.0.0-mymaria-mini-prometheus.yaml -n mariadb-system

kubectl run alpinecurl --rm --tty -i            \
            --restart='Never'                   \
            --image docker.io/alpine/curl:3.14  \
            --namespace mariadb-system          \
            --command -- curl http://mymaria-mariadb-primary.mariadb-system.svc.cluster.local:9104/metrics

grep -v '#' mariadb-metrics

kubectl run alpinecurl --rm --tty -i            \
            --restart='Never'                   \
            --image docker.io/alpine/curl:3.14  \
            --namespace mariadb-system          \
            --command -- curl http://mymaria-mariadb-primary.mariadb-system.svc.cluster.local:9104/metrics | \
grep --color=no 'mysql_global_status_memory_used' | grep -v '#'
